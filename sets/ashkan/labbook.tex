% Typography: the overfull boxes in this document are almost all long \texttt
% run ids and file paths, which TeX will not break by default. [htt] lets it
% hyphenate typewriter text; the emergency stretch absorbs the rest rather than
% letting a line cross the margin. Together with \pathid (labbook/macros.tex)
% this took the count from 49 to 8, worst case 165pt to 59pt.
%
% [htt] costs ~31 "Font shape T1/cmtt/... undefined" substitution warnings in
% the log. Kept anyway: without it the count is 17. Log noise is cheaper than
% text crossing the margin.
\usepackage[htt]{hyphenat}
\emergencystretch=3em
\hyphenpenalty=250
\tolerance=1500
% Run ids like 2026-07-29-lattice-band-validation are wider than the columns
% that hold them; breaking after their existing hyphens must be cheap.
\exhyphenpenalty=50
